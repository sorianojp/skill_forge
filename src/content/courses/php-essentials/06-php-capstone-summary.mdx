declare(strict_types=1);

function summarizeScores(array $scores): array {
    $count = count($scores);
    $passed = array_values(array_filter($scores, fn($s) => $s >= 80));
    $average = $count > 0 ? round(array_sum($scores) / $count, 1) : 0.0;

    return [
        'count' => $count,
        'passed' => $passed,
        'average' => $average,
    ];
}

echo json_encode(summarizeScores([74, 82, 96, 88])) . "\\n";`}
