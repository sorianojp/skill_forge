## Capstone Code Lab (Node.js 20)

<CodeLab
  client:visible
  title="task_api.js"
  language="node"
  code={`const crypto = require("node:crypto");

const db = [];

function createTask(body) {
  const title = typeof body?.title === "string" ? body.title.trim() : "";
  if (title.length < 3) {
    return { status: 400, body: { error: "Title must be at least 3 chars" } };
  }
  const task = { id: crypto.randomUUID().slice(0, 8), title, done: false };
  db.push(task);
  return { status: 201, body: task };
}

console.log(createTask({ title: "Hi" }));
console.log("Created status:", createTask({ title: "Deploy Node API" }).status, "| Count:", db.length);`}
  expectedOutput={`{ status: 400, body: { error: 'Title must be at least 3 chars' } }
Created status: 201 | Count: 1`}
/>
