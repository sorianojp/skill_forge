## ES Modules (`import` / `export`)

Every modern browser natively supports `<script type="module">`, letting you split your app into small, focused files:

```js
// math.js
export const clamp = (n, min, max) => Math.min(Math.max(n, min), max);
export default function double(n) { return n * 2; }

// app.js
import double, { clamp } from './math.js';
```

## Modern Classes & True Private Fields (`#`)

Prefixing a property with `#` makes it **genuinely private**, external code cannot read or mutate it directly:

```js
class ProgressTracker {
  #completed = new Set();

  complete(slug) { this.#completed.add(slug); }
  get count() { return this.#completed.size; }
}
```

<Callout type="tip" title="Module Scope Is Automatic">
  Variables declared at the top level of an ES Module stay private to that file unless explicitly exported, no more accidental global variable collisions!
</Callout>

  title="ES6 Class with Private #completed Set"
  defaultTab="js"

.out { margin-top: 12px; font-size: 14px; font-weight: 600; color: #0f172a; }`}
  js={`class CourseTracker {
  #completed = new Set();
  constructor(total) {
    this.total = total;
  }
  markDone(lesson) {
    this.#completed.add(lesson);
  }
  get summary() {
    const pct = Math.round((this.#completed.size / this.total) * 100);
    return \`Completed: \${this.#completed.size} / \${this.total} (\${pct}%)\`;
  }
}

const tracker = new CourseTracker(3);
const lessons = ['html-1', 'html-2', 'html-3'];
let idx = 0;

document.querySelector('#mark').addEventListener('click', () => {
  if (idx < lessons.length) {
    tracker.markDone(lessons[idx++]);
    document.querySelector('#out').textContent = tracker.summary;
    console.log(tracker.summary);
  }
});`}
