Most real web apps fetch a list of items (courses, products, messages) as an **array of objects**, filter or sort it, and render it into HTML.

## `.filter()` + `.map()` Pipeline

```js
const courses = [
  { title: 'HTML', free: true },
  { title: 'CSS', free: true },
  { title: 'Pro Workshop', free: false },
];

const freeTitles = courses
  .filter((c) => c.free)
  .map((c) => c.title);
// ['HTML', 'CSS']
```

<Callout type="tip" title="Immutability">
  Both <code>.filter()</code> and <code>.map()</code> return <strong>brand-new arrays</strong> without mutating the original data, making bugs much easier to avoid.
</Callout>

  { title: 'Modern HTML Fundamentals', level: 'Beginner' },
  { title: 'Modern CSS Layout', level: 'Beginner' },
  { title: 'Async JavaScript Patterns', level: 'Intermediate' },
  { title: 'Web Performance & CWV', level: 'Intermediate' },
];

const list = document.querySelector('#courseList');
const buttons = document.querySelectorAll('.filters button');

function render(level) {
  const visible = level === 'All'
    ? courses
    : courses.filter((c) => c.level === level);

  list.replaceChildren(...visible.map((c) => {
    const li = document.createElement('li');
    li.className = 'item';
    li.innerHTML = \`<strong>\${c.title}</strong><span class="tag">\${c.level}</span>\`;
    return li;
  }));

  console.log(\`Showing \${visible.length} course(s) for "\${level}"\`);
}

buttons.forEach((btn) => {
  btn.addEventListener('click', () => {
    buttons.forEach((b) => b.classList.remove('active'));
    btn.classList.add('active');
    render(btn.dataset.level);
  });
});

render('All');`}
