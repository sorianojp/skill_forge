## Try It Yourself: Task Tracker

Add a task, click a task to toggle completion, or click `×` to remove it:

<Playground
  client:visible
  title="JavaScript Capstone: Task Tracker"

  showConsole={true}

  html={`<div class="tracker">
  <form id="taskForm" class="row">
    <input id="taskInput" type="text" placeholder="Add a study goal..." required />
    <button type="submit">Add</button>
  </form>
  <p id="summary" class="summary"></p>
  <ul id="taskList" class="list"></ul>
</div>`}
  css={`.tracker { max-width: 380px; }
.row { display: flex; gap: 8px; }
input { flex: 1; padding: 8px 10px; border: 1px solid #cbd5e1; border-radius: 8px; font-size: 13px; }
button { background: #2563eb; color: #fff; border: none; border-radius: 8px; padding: 8px 14px; font-weight: 600; cursor: pointer; }
.summary { font-size: 12px; color: #64748b; margin: 10px 0; font-weight: 600; }
.list { list-style: none; padding: 0; margin: 0; display: grid; gap: 6px; }
.task { display: flex; align-items: center; justify-content: space-between; gap: 8px; padding: 8px 10px; border: 1px solid #e2e8f0; border-radius: 8px; background: #f8fafc; font-size: 13px; }
.task.done span { text-decoration: line-through; color: #94a3b8; }
.toggle { background: none; color: inherit; padding: 0; text-align: left; flex: 1; font-weight: 500; }
.del { background: #fee2e2; color: #b91c1c; padding: 2px 8px; font-size: 12px; }`}
  js={`let tasks = [
  { id: 1, text: 'Finish HTML Fundamentals', done: true },
  { id: 2, text: 'Build CSS Grid dashboard', done: false },
];

const form = document.querySelector('#taskForm');
const input = document.querySelector('#taskInput');
const list = document.querySelector('#taskList');
const summary = document.querySelector('#summary');

function render() {
  const remaining = tasks.filter((t) => !t.done).length;
  summary.textContent = \`\${remaining} of \${tasks.length} tasks remaining\`;

  list.replaceChildren(...tasks.map((t) => {
    const li = document.createElement('li');
    li.className = 'task' + (t.done ? ' done' : '');
    li.innerHTML = \`
      <button type="button" class="toggle" data-action="toggle" data-id="\${t.id}">
        <span>\${t.done ? '✓ ' : '○ '}\${t.text}</span>
      </button>
      <button type="button" class="del" data-action="delete" data-id="\${t.id}" aria-label="Delete">×</button>
    \`;
    return li;
  }));
}

form.addEventListener('submit', (e) => {
  e.preventDefault();
  const text = input.value.trim();
  if (!text) return;
  tasks.push({ id: Date.now(), text, done: false });
  input.value = '';
  console.log('Added task:', text);
  render();
});

list.addEventListener('click', (e) => {
  const btn = e.target.closest('button[data-action]');
  if (!btn) return;
  const id = Number(btn.dataset.id);
  if (btn.dataset.action === 'toggle') {
    tasks = tasks.map((t) => (t.id === id ? { ...t, done: !t.done } : t));
  } else if (btn.dataset.action === 'delete') {
    tasks = tasks.filter((t) => t.id !== id);
  }
  render();
});

render();`}
/>
