## Capstone Code Lab

<CodeLab
  client:visible
  title="CatalogService.cs"
  language="csharp"
  code={`using System.Linq;

public record Track(string Name, int Lessons, bool Published);

public interface ITrackRepo
{
    Task<List<Track>> GetAllAsync();
}

public class InMemoryTrackRepo : ITrackRepo
{
    public Task<List<Track>> GetAllAsync() => Task.FromResult(new List<Track>
    {
        new("Git", 6, true),
        new("OOP", 6, true),
        new("C#", 6, true),
        new("Draft", 2, false)
    });
}

ITrackRepo repo = new InMemoryTrackRepo();
var live = (await repo.GetAllAsync()).Where(t => t.Published).ToList();
Console.WriteLine($"Live tracks: {live.Count} | Lessons: {live.Sum(t => t.Lessons)}");`}
  expectedOutput={`Live tracks: 3 | Lessons: 18`}
/>
