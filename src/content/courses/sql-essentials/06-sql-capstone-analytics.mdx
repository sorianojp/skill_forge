CREATE TABLE enrollments (id INTEGER PRIMARY KEY, course_id INTEGER, score INTEGER);
INSERT INTO courses VALUES (1, 'HTML'), (2, 'CSS'), (3, 'SQL');

  (1, 1, 88), (2, 1, 92),
  (3, 2, 80), (4, 2, 86),

  query={`SELECT
  c.title,
  COUNT(e.id) AS learner_count,
  ROUND(AVG(e.score), 1) AS avg_score
FROM courses c
INNER JOIN enrollments e ON e.course_id = c.id
GROUP BY c.id, c.title
ORDER BY avg_score DESC;`}
