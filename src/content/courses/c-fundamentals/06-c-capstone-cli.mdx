## Capstone Code Lab

<CodeLab
  client:visible
  title="analyzer.c"
  language="c"
  code={`#include <stdio.h>

typedef struct {
    const char *label;
    int score;
} Entry;

void summarize(const Entry *items, size_t n, double *avg, int *top) {
    int sum = 0;
    int best = items[0].score;
    for (size_t i = 0; i < n; i++) {
        sum += items[i].score;
        if (items[i].score > best) best = items[i].score;
    }
    *avg = (double)sum / n;
    *top = best;
}

int main(void) {
    Entry data[] = { {"Quiz 1", 85}, {"Quiz 2", 94}, {"Exam", 91} };
    double avg = 0.0;
    int top = 0;

    summarize(data, 3, &avg, &top);
    printf("Average: %.1f | Top: %d\\n", avg, top);
    return 0;
}`}
  expectedOutput={`Average: 90.0 | Top: 94`}
/>
