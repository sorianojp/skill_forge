## Capstone Bash Lab

<CodeLab
  client:visible
  title="log_analyzer.sh"
  language="bash"
  code={`#!/usr/bin/env bash
set -euo pipefail

logs=$(cat << 'EOF'
/ 200
/courses 200
/missing 404
/api/run 200
/old 404
EOF
)

echo "Status summary:"
echo "$logs" | awk '{print $2}' | sort | uniq -c | awk '{print "HTTP " $2 ": " $1 " req(s)"}'`}
  expectedOutput={`Status summary:
HTTP 200: 3 req(s)
HTTP 404: 2 req(s)`}
/>
