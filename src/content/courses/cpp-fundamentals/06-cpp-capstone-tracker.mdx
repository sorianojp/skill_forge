## Capstone Code Lab

<CodeLab
  client:visible
  title="inventory.cpp"
  language="cpp"
  code={`#include <iostream>
#include <string>
#include <vector>

struct Item {
    std::string name;
    double price;
    int qty;
};

class Inventory {
    std::vector<Item> items_;
public:
    void add(const Item& item) {
        items_.push_back(item);
    }

    double total_value() const {
        double total = 0.0;
        for (const auto& item : items_) {
            total += item.price * item.qty;
        }
        return total;
    }
};

int main() {
    Inventory inv;
    inv.add({"Keyboard", 85.0, 2});
    inv.add({"Mouse", 40.0, 3});
    std::cout << "Total inventory value: $" << inv.total_value() << "\\n";
    return 0;
}`}
  expectedOutput={`Total inventory value: $290`}
/>
