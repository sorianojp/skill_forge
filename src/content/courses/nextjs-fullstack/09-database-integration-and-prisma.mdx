## Production Code Example

```tsx
// lib/db.ts (singleton pattern for Next.js dev HMR)
const globalForDb = globalThis as unknown as { dbPool?: unknown };

export function getDbPool() {
  if (!globalForDb.dbPool) {
    globalForDb.dbPool = { connected: true };
  }
  return globalForDb.dbPool;
}
```
