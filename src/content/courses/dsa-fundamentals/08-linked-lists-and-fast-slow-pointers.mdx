Two classic pointer techniques solve 80% of linked list interview problems:
1. **In-place reversal:** Save `next = curr.next`, flip `curr.next = prev`, advance `prev = curr; curr = next`.
2. **Fast & Slow pointers (Floyd's Tortoise & Hare):** Move `slow` by 1 step and `fast` by 2 steps. When `fast` reaches the end, `slow` is at the exact **middle**; if there is a **cycle**, `fast` catches up to `slow` in $O(1)$ space!

<Callout type="tip" title="Use a Dummy Head Node">
  When inserting or deleting nodes that might touch the head of a list, creating a temporary <code>dummy = &#123; next: head &#125;</code> removes special-case edge checks.
</Callout>

  js={`let head = { val: 1, next: { val: 2, next: { val: 3, next: { val: 4, next: null } } } };

function reverse(node) {
  let prev = null, curr = node;
  while (curr) {
    const next = curr.next;
    curr.next = prev;
    prev = curr;
    curr = next;
  }
  return prev;
}

function toArray(node) {
  const out = [];
  for (let c = node; c; c = c.next) out.push(c.val);
  return out;
}

console.log("Before:", toArray(head));
console.log("After: ", toArray(reverse(head)));`}
