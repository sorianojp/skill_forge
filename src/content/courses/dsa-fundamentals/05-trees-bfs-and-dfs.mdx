- **DFS (Depth-First Search):** Explores as deep as possible along each branch before backtracking (uses recursion or a Stack). Great for computing tree height, path sums, and topological sort.
- **BFS (Breadth-First Search):** Explores level by level (uses a Queue). Guarantees the **shortest path** (fewest edges) in unweighted graphs.

<Callout type="tip" title="Remember the Queue vs. Stack Rule">
  BFS uses a <strong>Queue</strong> (FIFO); DFS uses a <strong>Stack</strong> (or recursion call stack).
</Callout>

  val: 10,
  left: { val: 5, left: { val: 2 }, right: { val: 7 } },
  right: { val: 15, right: { val: 20 } }
};

function bfs(node) {
  const queue = [node], order = [];
  for (let head = 0; head < queue.length; head++) {
    const cur = queue[head];
    order.push(cur.val);
    if (cur.left) queue.push(cur.left);
    if (cur.right) queue.push(cur.right);
  }
  return order;
}

function dfsSum(node) {
  if (!node) return 0;
  return node.val + dfsSum(node.left) + dfsSum(node.right);
}

console.log("BFS order:", bfs(root));
console.log("DFS sum:  ", dfsSum(root));`}
