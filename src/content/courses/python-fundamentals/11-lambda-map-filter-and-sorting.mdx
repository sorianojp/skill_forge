  title="sorted(), zip(), any() & all()"

scores = [92, 85, 98]

leaderboard = sorted(zip(names, scores), key=lambda pair: pair[1], reverse=True)
for rank, (name, score) in enumerate(leaderboard, start=1):
    print(f"#{rank} {name}: {score}%")

print("All passed (>=80):", all(s >= 80 for s in scores))
print("Any perfect (100):", any(s == 100 for s in scores))`}
