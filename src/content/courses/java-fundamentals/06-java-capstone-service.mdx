## Capstone Code Lab

<CodeLab
  client:visible
  title="OrderService.java"
  language="java"
  code={`import java.util.List;

public class OrderService {
    record OrderItem(String name, double price, int qty) {
        double subtotal() {
            return price * qty;
        }
    }

    public static void main(String[] args) {
        var items = List.of(
            new OrderItem("Mechanical Keyboard", 95.0, 1),
            new OrderItem("USB-C Cable", 12.5, 2)
        );

        double total = items.stream()
            .mapToDouble(OrderItem::subtotal)
            .sum();

        System.out.printf("Items: %d | Total: $%.2f%n", items.size(), total);
    }
}`}
  expectedOutput={`Items: 2 | Total: $120.00`}
/>
