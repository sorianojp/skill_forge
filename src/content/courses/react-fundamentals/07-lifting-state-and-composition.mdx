When two sibling components (like `<SearchBar />` and `<CourseList />`) need to stay in sync, **lift the state up** to their nearest common parent (`<CatalogPage />`). The parent owns `query`, passes `query` and `onChange` to `<SearchBar />`, and passes the filtered list to `<CourseList />`.

<Callout type="tip" title="Composition via children Beats Prop Drilling">
  Instead of passing <code>user</code> through 4 layout wrappers, pass <code>&lt;Header user=&#123;user&#125; /&gt;</code> directly as <code>children</code> into <code>&lt;PageShell&gt;</code>.
</Callout>

function render() {
  document.getElementById("panel").textContent = "Tracks: " + data[activeTab];
}

document.querySelectorAll("button[data-tab]").forEach((btn) => {
  btn.addEventListener("click", () => {
    activeTab = btn.dataset.tab;
    render();
  });
});
render();`}
