## Capstone Playground

<ReactPlayground
  client:visible
  title="Interactive React Task Board (Live JSX)"
  height={290}
  code={`function App() {
  const [tasks, setTasks] = useState([
    { id: 1, title: 'Learn Components & Props', done: true },
    { id: 2, title: 'Master Immutable State', done: false },
    { id: 3, title: 'Pass React Final Exam', done: false },
  ]);

  const toggle = (id) =>
    setTasks((prev) => prev.map((t) => (t.id === id ? { ...t, done: !t.done } : t)));

  const remaining = tasks.filter((t) => !t.done).length;

  return (
    <div>
      <p style={{ fontWeight: 700, margin: '0 0 8px' }}>
        {remaining} remaining of {tasks.length}
      </p>
      <div style={{ display: 'grid', gap: 6 }}>
        {tasks.map((t) => (
          <button
            key={t.id}
            onClick={() => toggle(t.id)}
            style={{
              textAlign: 'left',
              padding: '8px 12px',
              borderRadius: 10,
              border: 0,
              background: t.done ? '#dcfce7' : '#f1f5f9',
            }}
          >
            {t.done ? '✓ ' : '○ '}{t.title}
          </button>
        ))}
      </div>
    </div>
  );
}`}
/>
