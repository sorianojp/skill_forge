## Capstone Code Lab

<CodeLab
  client:visible
  title="store.ts"
  language="ts"
  code={`interface CounterState {
  readonly count: number;
}

type Action =
  | { type: "inc"; by: number }
  | { type: "reset" };

function reducer(state: CounterState, action: Action): CounterState {
  switch (action.type) {
    case "inc": return { count: state.count + action.by };
    case "reset": return { count: 0 };
  }
}

let state: CounterState = { count: 0 };
state = reducer(state, { type: "inc", by: 5 });
state = reducer(state, { type: "inc", by: 3 });
console.log("Final count:", state.count);`}
  expectedOutput={`Final count: 8`}
/>
